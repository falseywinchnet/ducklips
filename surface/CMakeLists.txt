cmake_minimum_required(VERSION 3.24)
project(DuckLipsSurface LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
add_library(ducklips_surface STATIC surface_network.cpp)
target_include_directories(ducklips_surface PUBLIC .)
target_compile_options(ducklips_surface PRIVATE -Wall -Wextra -Wpedantic)
add_executable(surface_probe surface_probe.cpp)
target_link_libraries(surface_probe PRIVATE ducklips_surface)
add_subdirectory(.. ducklips_core EXCLUDE_FROM_ALL)
add_executable(surface_areas surface_areas.cpp)
target_link_libraries(surface_areas PRIVATE ducklips)
add_executable(surface_render surface_render.cpp)
target_link_libraries(surface_render PRIVATE ducklips_surface ducklips)
add_library(ducklips_surface_synth STATIC surface_synthesizer.cpp)
target_include_directories(ducklips_surface_synth PUBLIC .)
target_compile_options(ducklips_surface_synth PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ducklips_surface_synth PUBLIC ducklips_surface ducklips)
add_executable(surface_say surface_say.cpp)
target_link_libraries(surface_say PRIVATE ducklips_surface_synth)

add_executable(surface_response surface_response.cpp)
target_link_libraries(surface_response PRIVATE ducklips_surface_synth)
add_executable(surface_linear_tests surface_linear_tests.cpp)
target_link_libraries(surface_linear_tests PRIVATE ducklips_surface)
# The parent project registers these with CTest even when its other tools are
# excluded from the surface build. Keep the registered test executables built.
add_dependencies(surface_linear_tests ducklips_tests ducklips_stream_tests ducklips_moving_mode_tests)
enable_testing()
add_test(NAME surface_linear_tests COMMAND surface_linear_tests)

add_library(ducklips_frozen_modes STATIC frozen_modes.cpp)
target_include_directories(ducklips_frozen_modes PUBLIC .)
target_compile_options(ducklips_frozen_modes PRIVATE -Wall -Wextra -Wpedantic)
add_executable(surface_modal_probe surface_modal_probe.cpp)
target_link_libraries(surface_modal_probe PRIVATE ducklips_frozen_modes ducklips)
add_executable(surface_modal_tests frozen_modes_tests.cpp)
target_link_libraries(surface_modal_tests PRIVATE ducklips_frozen_modes)
add_test(NAME surface_modal_tests COMMAND surface_modal_tests)
